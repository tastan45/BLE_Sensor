// v2.2 - Güncellenmiş BLE Dashboard Kodu (ATC 4-Byte Payload ve Notify Desteği) async function connectAndSubscribe() { try { const serviceUuid = 'ebe0ccc0-75a3-4f51-9f8e-d9036c6460fa'; const characteristicUuid = 'ebe0ccc1-75a3-4f51-9f8e-d9036c6460fa'; console.log('Bluetooth cihazı aranıyor...'); const device = await navigator.bluetooth.requestDevice({ filters: [{ services: [serviceUuid] }], optionalServices: [serviceUuid] }); const server = await device.gatt.connect(); const service = await server.getPrimaryService(serviceUuid); const characteristic = await service.getCharacteristic(characteristicUuid); // Bildirimleri başlat (Sürekli veri akışı için kritik) await characteristic.startNotifications(); characteristic.addEventListener('characteristicvaluechanged', (event) => { const value = event.target.value; // ATC custom firmware 4-byte payload formatı: [Temp(2) | Hum(1) | Batt(1)] if (value.byteLength >= 4) { const tempRaw = value.getInt16(0, true); // Little-endian const temperature = tempRaw / 100; const humidity = value.getUint8(2); const battery = value.getUint8(3); console.log(`Sıcaklık: ${temperature}°C, Nem: %${humidity}, Pil: %${battery}`); // Arayüz güncellemelerini burada yapabilirsiniz document.getElementById('temp').innerText = `${temperature} °C`; document.getElementById('humidity').innerText = `% ${humidity}`; document.getElementById('battery').innerText = `% ${battery}`; } else { // Tek bayt gelmesi durumunda (örneğin sadece pil veya meta veri) const singleByte = value.getUint8(0); console.log(`Tek bayt alındı (Muhtemel Pil): ${singleByte}`); document.getElementById('battery').innerText = `% ${singleByte}`; } }); console.log('Bildirimler aktif, veri bekleniyor...'); } catch (error) { console.error('BLE bağlantı hatası:', error); } }
