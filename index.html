// v2.3 - BLE Tarama Verisi (Advertisement Data) ile Doğrudan Okuma async function startScanForATC() { try { console.log('Bluetooth reklam paketleri taranıyor...'); const options = { acceptAllAdvertisements: true, optionalServices: ['ebe0ccc0-75a3-4f51-9f8e-d9036c6460fa'] }; const scanner = await navigator.bluetooth.requestLEScan(options); navigator.bluetooth.addEventListener('advertisementreceived', (event) => { const device = event.device; const manufacturerData = event.manufacturerData; // ATC custom firmware genellikle Manufacturer Specific Data içinde ver yayınlar for (const [companyId, data] of manufacturerData.entries()) { // ATC formatı kontrolü veya veri uzunluğu analizi (Genellikle 13+ bayt olur) if (data.byteLength >= 13) { const value = new DataView(data.buffer); // ATC formatına göre ayrıştırma (örnek offsetler) const tempRaw = value.getInt16(6, true); // Sıcaklık offset const temperature = tempRaw / 100; const humidity = value.getUint8(8); // Nem offset const battery = value.getUint8(9); // Pil offset console.log(`[Scan] Cihaz: ${device.name}, Sıcaklık: ${temperature}°C, Nem: %${humidity}, Pil: %${battery}`); document.getElementById('temp').innerText = `${temperature} °C`; document.getElementById('humidity').innerText = `% ${humidity}`; document.getElementById('battery').innerText = `% ${battery}`; } } }); console.log('Reklam dinleme başlatıldı...'); } catch (error) { console.error('Tarama hatası (HTTPS veya flag gerekebilir):', error); } }
